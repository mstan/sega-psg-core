cmake_minimum_required(VERSION 3.20)
project(sega_psg_core VERSION 0.1.0 LANGUAGES C)
option(SEGA_PSG_BUILD_TESTS "Build independent donor fixtures" ON)
add_library(sega_psg_core STATIC src/sega_psg.c src/genesis.c src/sms.c)
add_library(sega-psg::core ALIAS sega_psg_core)
target_include_directories(sega_psg_core PUBLIC include)
target_compile_features(sega_psg_core PUBLIC c_std_99)
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
 target_compile_options(sega_psg_core PRIVATE -Wall -Wextra -Wno-unused-parameter -Wno-unused-function)
endif()
if(SEGA_PSG_BUILD_TESTS)
 enable_testing()
 add_executable(psg_differential tests/psg_test.c tests/oracle_genesis.c tests/oracle_sms.c)
 target_link_libraries(psg_differential PRIVATE sega-psg::core)
 add_test(NAME donor_differential COMMAND psg_differential)
endif()
